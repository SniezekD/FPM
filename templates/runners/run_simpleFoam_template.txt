#!/bin/bash

check_return_code () {
if [ $1 -eq 0 ]; then
    echo "     Done"
else
    echo "     Error with code $1, exiting."
    exit 1
fi
}

source /usr/lib/openfoam/openfoam2006/etc/bashrc

cd ../wd/OF_Model 
echo "    Decomposing"
decomposePar -force &> decomposePar.log
check_return_code $?

echo "    Running {{ solver_of_name }}"
if [ ! -d {{ solver_of_name }}Logs ]; then
    mkdir {{ solver_of_name }}Logs
fi

mpirun -np {{ number_of_procs }}  {{ solver_of_name }} -parallel &>  {{ solver_of_name }}Logs/ {{ solver_of_name }}{{ reynolds_number }}.log

check_return_code $?

echo "    Reconstructing"
reconstructPar -latestTime &> reconstructPar.log
check_return_code $?

echo "    Converting foam to VTK"
foamToVTK -latestTime -ascii  &> foamToVTK.log
check_return_code $?

echo "    Removing all processors' directories"
rm -r processor*
check_return_code $?
